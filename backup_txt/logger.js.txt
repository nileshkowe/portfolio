const fs = typeof window === 'undefined' ? require('fs') : null; // Only require fs on the server
const path = typeof window === 'undefined' ? require('path') : null; // Only require path on the server

let logDir;
if (fs && path) {
  logDir = path.join(__dirname, 'logs');
  // Ensure logs directory only if fs is available
  // Check if directory exists before creating
  if (!fs.existsSync(logDir)) {
    try {
      fs.mkdirSync(logDir, { recursive: true }); // Added recursive to ensure parent dirs are created if needed
    } catch (err) {
      // In some environments (like Vercel build), __dirname might not be where you expect,
      // or file system might be read-only. Log error to console.
      console.error('Failed to create log directory:', err);
      logDir = null; // Disable file logging if dir creation fails
    }
  }
} else {
  logDir = null; // Disable file logging on the client or if path/fs failed
}

/**
 * @returns {string} Current timestamp in ISO format.
 */
function timestamp() {
  return new Date().toISOString();
}

/**
 * Writes a log message to the appropriate log file if fs is available.
 * @param {string} level - The log level (e.g., 'info', 'error').
 * @param {string} msg - The message to log.
 */
function writeToFile(level, msg) {
  if (!fs || !path || !logDir) { // Check if fs, path, and logDir are available
    return; // Do nothing on the client-side or if setup failed
  }
  const line = `[${timestamp()}] [${level.toUpperCase()}] ${msg}\n`;
  try {
    fs.appendFileSync(path.join(logDir, `${level}.log`), line);
  } catch (err) {
    console.error(`Failed to write to ${level}.log:`, err);
  }
}

/**
 * Logger module with different logging levels.
 * Logs to console and attempts to write to file on server-side.
 */
module.exports = {
  /** @param {string} msg - Message to log. */
  trace: msg => { console.debug(`[TRACE] ${msg}`); writeToFile('trace', msg); },
  /** @param {string} msg - Message to log. */
  debug: msg => { console.debug(`[DEBUG] ${msg}`); writeToFile('debug', msg); },
  /** @param {string} msg - Message to log. */
  info:  msg => { console.log (`[INFO ] ${msg}`); writeToFile('info',  msg); },
  /** @param {string} msg - Message to log. */
  warn:  msg => { console.warn(`[WARN ] ${msg}`); writeToFile('warn',  msg); },
  /** @param {string} msg - Message to log. */
  error: msg => { console.error(`[ERROR] ${msg}`); writeToFile('error', msg); },
};
