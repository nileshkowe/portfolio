/**
 * @file Logger utility for the portfolio website.
 * Provides timestamped logging to both console and files.
 */

// Node.js core modules - dynamically imported for server-side only
import type fsModule from 'fs';
import type pathModule from 'path';

let fs: typeof fsModule | null = null;
let path: typeof pathModule | null = null;
let logDir: string | null = null;

/**
 * Flag to determine if the code is running on the server.
 */
const IS_SERVER = typeof window === 'undefined';

let ensureLogDirPromise: Promise<void> | null = null;

if (IS_SERVER) {
  // Dynamically require 'fs' and 'path' only on the server
  // to prevent Webpack from trying to bundle them for the client.
  const fsPromise = import('fs').then(fsImport => fs = fsImport.default).catch(err => console.error('[ERROR] Failed to load fs module:', err));
  const pathPromise = import('path').then(pathImport => path = pathImport.default).catch(err => console.error('[ERROR] Failed to load path module:', err));

  ensureLogDirPromise = Promise.all([fsPromise, pathPromise]).then(() => {
    if (fs && path) {
      logDir = path.join(process.cwd(), 'logs');
      if (!fs.existsSync(logDir)) {
        try {
          fs.mkdirSync(logDir, { recursive: true });
          // console.log(`[INFO ] Log directory created at: ${logDir}`); // Optional: confirm dir creation
        } catch (err) {
          console.error('[ERROR] Failed to create log directory asynchronously:', err);
          logDir = null; // Reset logDir if creation failed
        }
      } else {
        // console.log(`[INFO ] Log directory already exists at: ${logDir}`); // Optional: confirm dir exists
      }
    } else {
      console.error('[ERROR] fs or path module not loaded, cannot ensure log directory.');
    }
  }).catch(err => {
    console.error('[ERROR] Error during dynamic import of fs/path for logger setup:', err);
  });
}

/**
 * Generates a timestamp string in ISO format.
 * @returns {string} The current timestamp.
 */
function getTimestamp(): string {
  return new Date().toISOString();
}

/**
 * Writes a log message to the specified log file.
 * This function will only operate on the server-side.
 * @param {string} level - The log level (e.g., 'TRACE', 'DEBUG').
 * @param {string} message - The log message.
 */
async function writeLogToFile(level: string, message: string): Promise<void> {
  if (!IS_SERVER) return;

  if (ensureLogDirPromise) {
    await ensureLogDirPromise; // Ensure directory setup is complete
  }

  if (fs && path && logDir) {
    const logFilePath = path.join(logDir, `${level.toLowerCase()}.log`);
    const logEntry = `[${getTimestamp()}] [${level.toUpperCase()}] ${message}\n`;
    try {
      fs.appendFileSync(logFilePath, logEntry);
    } catch (err) {
      // Fallback to console.error if file logging fails on server
      console.error(`[ERROR] Server-side logger failed to write to ${level.toLowerCase()}.log:`, err);
    }
  } else if (IS_SERVER) { // Removed condition (!fs || !path || !logDir) as it's implicitly covered
    // This warning is more relevant if ensureLogDirPromise failed or modules are truly missing
    console.warn(`[WARN] Server-side logger: fs, path, or logDir not available. Cannot write to ${level.toLowerCase()}.log. Message: ${message}`);
  }
}

/**
 * Logger module with different logging levels.
 */
export const logger = {
  /**
   * Logs a trace message.
   * Used for highly detailed, step-by-step information.
   * @param {string} message - The message to log.
   */
  trace: (message: string): void => {
    console.debug(`[TRACE] ${message}`);
    if (IS_SERVER) {
      writeLogToFile('trace', message).catch(err => console.error("[ERROR] Async trace file write failed:", err));
    }
  },
  /**
   * Logs a debug message.
   * Used for function entry/exit, key variable states.
   * @param {string} message - The message to log.
   */
  debug: (message: string): void => {
    console.debug(`[DEBUG] ${message}`);
    if (IS_SERVER) {
      writeLogToFile('debug', message).catch(err => console.error("[ERROR] Async debug file write failed:", err));
    }
  },
  /**
   * Logs an info message.
   * Used for high-level milestones (e.g., server start, DB connected).
   * @param {string} message - The message to log.
   */
  info: (message: string): void => {
    console.log(`[INFO ] ${message}`);
    if (IS_SERVER) {
      writeLogToFile('info', message).catch(err => console.error("[ERROR] Async info file write failed:", err));
    }
  },
  /**
   * Logs a warning message.
   * Used for recoverable issues or potential problems.
   * @param {string} message - The message to log.
   */
  warn: (message: string): void => {
    console.warn(`[WARN ] ${message}`);
    if (IS_SERVER) {
      writeLogToFile('warn', message).catch(err => console.error("[ERROR] Async warn file write failed:", err));
    }
  },
  /**
   * Logs an error message.
   * Used for uncaught exceptions, failures, or critical errors.
   * @param {string} message - The message to log.
   * @param {any} [errorObject] - Optional error object to log.
   */
  error: (message: string, errorObject?: any): void => {
    console.error(`[ERROR] ${message}`, errorObject || '');
    if (IS_SERVER) {
      let fullMessage = message;
      if (errorObject) {
        fullMessage += ` | ${errorObject instanceof Error ? errorObject.stack || errorObject.message : JSON.stringify(errorObject)}`;
      }
      writeLogToFile('error', fullMessage).catch(err => console.error("[ERROR] Async error file write failed:", err));
    }
  },
};

// Export individual functions for easier import if preferred
// These will also benefit from the IS_SERVER check within the logger object's methods.
export const { trace, debug, info, warn, error } = logger;
